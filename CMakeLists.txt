cmake_minimum_required(VERSION 3.16)
project(Tumpuk LANGUAGES CXX)

if(WIN32)
    set(DEFAULT_FREEGLUT OFF)
else()
    set(DEFAULT_FREEGLUT ON)
endif()
option(TUMPUK_FREEGLUT "Use the freeglut window backend" ${DEFAULT_FREEGLUT})

find_package(OpenGL REQUIRED)
if(TUMPUK_FREEGLUT)
    find_package(GLUT REQUIRED)
    add_executable(Tumpuk src/app.cpp src/main_freeglut.cpp)
    target_link_libraries(Tumpuk PRIVATE OpenGL::GL GLUT::GLUT)
elseif(WIN32)
    add_executable(Tumpuk WIN32 src/app.cpp src/main_win32.cpp)
    target_link_libraries(Tumpuk PRIVATE OpenGL::GL user32 gdi32)
    if(MINGW)
        target_link_options(Tumpuk PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()
else()
    message(FATAL_ERROR "Use TUMPUK_FREEGLUT=ON on Linux.")
endif()
target_compile_features(Tumpuk PRIVATE cxx_std_17)
if(MSVC)
    set_property(TARGET Tumpuk PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(Tumpuk PRIVATE /W4 /utf-8 /EHsc)
else()
    target_compile_options(Tumpuk PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_custom_command(TARGET Tumpuk POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different
        "${CMAKE_CURRENT_SOURCE_DIR}/credits.txt" "$<TARGET_FILE_DIR:Tumpuk>/credits.txt")
